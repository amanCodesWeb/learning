# Aggregate Functions Deep Dive

`COUNT`, `SUM`, `AVG`, `MIN`, `MAX` — plus advanced variants, conditional aggregation, and window functions.

---

## COUNT — Counting Rows & Values

| Syntax | Counts |
|--------|--------|
| `COUNT(*)` | All rows (including NULLs) |
| `COUNT(col)` | Non-NULL values in column |
| `COUNT(DISTINCT col)` | Unique non-NULL values |
| `COUNT(DISTINCT col1, col2)` | Unique combinations |

```sql
-- Total rows
SELECT COUNT(*) FROM users;

-- Non-null emails
SELECT COUNT(email) FROM users;

-- Unique cities
SELECT COUNT(DISTINCT city) FROM users;

-- Unique (city, country) pairs
SELECT COUNT(DISTINCT city, country) FROM users;

-- Conditional count (MySQL: SUM(condition), PostgreSQL: COUNT(*) FILTER)
SELECT
    COUNT(*) AS total
    , SUM(is_active) AS active_count           -- MySQL: TRUE=1
    , COUNT(CASE WHEN is_active THEN 1 END) AS active_count  -- Standard
    , COUNT(*) FILTER (WHERE is_active) AS active_count      -- PostgreSQL
FROM users;
```

## SUM — Adding Values

```sql
-- Basic sum
SELECT SUM(total) FROM orders;

-- Sum with condition
SELECT
    SUM(total) AS all_revenue
    , SUM(CASE WHEN status = 'delivered' THEN total END) AS delivered_revenue
    , SUM(total) FILTER (WHERE status = 'delivered') AS delivered_revenue  -- PG
FROM orders;

-- Sum of computed expression
SELECT SUM(price * quantity) FROM order_items;

-- Sum with COALESCE (handle NULL)
SELECT COALESCE(SUM(total), 0) FROM orders WHERE status = 'impossible';
```

## AVG — Average

```sql
-- Simple average
SELECT AVG(total) FROM orders;

-- AVG ignores NULLs automatically
-- AVG(10, 20, NULL) = 15 (not 10)

-- Weighted average
SELECT SUM(price * quantity) / SUM(quantity) AS weighted_avg_price
FROM order_items;

-- Average with condition
SELECT AVG(CASE WHEN status = 'delivered' THEN total END) FROM orders;
```

## MIN / MAX — Extremes

```sql
-- Works on numbers, dates, strings
SELECT
    MIN(price) AS cheapest
    , MAX(price) AS most_expensive
    , MIN(created_at) AS first_order
    , MAX(created_at) AS last_order
    , MIN(name) AS first_alphabetically
    , MAX(name) AS last_alphabetically
FROM products;

-- MIN/MAX with condition
SELECT
    MIN(CASE WHEN status = 'delivered' THEN total END) AS smallest_delivered
    , MAX(CASE WHEN status = 'delivered' THEN total END) AS largest_delivered
FROM orders;
```

## Conditional Aggregation — The Power Pattern

Instead of multiple queries, do it all in one pass:

```sql
-- Order status breakdown
SELECT
    user_id
    , COUNT(*) AS total_orders
    , SUM(CASE WHEN status = 'pending' THEN 1 ELSE 0 END) AS pending
    , SUM(CASE WHEN status = 'paid' THEN 1 ELSE 0 END) AS paid
    , SUM(CASE WHEN status = 'shipped' THEN 1 ELSE 0 END) AS shipped
    , SUM(CASE WHEN status = 'delivered' THEN 1 ELSE 0 END) AS delivered
    , SUM(CASE WHEN status = 'cancelled' THEN 1 ELSE 0 END) AS cancelled
FROM orders
GROUP BY user_id;

-- Revenue by status
SELECT
    user_id
    , SUM(CASE WHEN status = 'delivered' THEN total ELSE 0 END) AS delivered_revenue
    , SUM(CASE WHEN status = 'cancelled' THEN total ELSE 0 END) AS cancelled_revenue
    , SUM(total) AS total_revenue
FROM orders
GROUP BY user_id;
```

### PostgreSQL: FILTER Clause (Cleaner)

```sql
SELECT
    user_id
    , COUNT(*) FILTER (WHERE status = 'pending') AS pending
    , COUNT(*) FILTER (WHERE status = 'delivered') AS delivered
    , SUM(total) FILTER (WHERE status = 'delivered') AS delivered_revenue
FROM orders
GROUP BY user_id;
```

### MySQL: Boolean Arithmetic

```sql
-- TRUE=1, FALSE=0 in numeric context
SELECT
    user_id
    , SUM(status = 'pending') AS pending
    , SUM(status = 'delivered') AS delivered
    , SUM((status = 'delivered') * total) AS delivered_revenue
FROM orders
GROUP BY user_id;
```

## Statistical Aggregates (PostgreSQL)

```sql
SELECT
    STDDEV_POP(price) AS pop_stddev      -- Population
    , STDDEV_SAMP(price) AS samp_stddev  -- Sample
    , VAR_POP(price) AS pop_variance
    , VAR_SAMP(price) AS samp_variance
    , PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY price) AS median
    , PERCENTILE_CONT(0.95) WITHIN GROUP (ORDER BY price) AS p95
    , PERCENTILE_DISC(0.5) WITHIN GROUP (ORDER BY price) AS median_discrete
FROM products;
```

## String Aggregation

```sql
-- MySQL
SELECT category, GROUP_CONCAT(name ORDER BY name SEPARATOR ', ') AS products
FROM products GROUP BY category;

-- PostgreSQL
SELECT category, STRING_AGG(name, ', ' ORDER BY name) AS products
FROM products GROUP BY category;

-- SQL Server
SELECT category, STRING_AGG(name, ', ') WITHIN GROUP (ORDER BY name) AS products
FROM products GROUP BY category;

-- Limit length (MySQL)
GROUP_CONCAT(name ORDER BY name SEPARATOR ', ' LIMIT 1000)
```

## JSON Aggregation (PostgreSQL / MySQL 5.7+)

```sql
-- PostgreSQL: JSON array of objects
SELECT
    user_id
    , JSON_AGG(
        JSON_BUILD_OBJECT(
            'order_id', id,
            'total', total,
            'status', status,
            'date', created_at
        ) ORDER BY created_at DESC
    ) AS orders
FROM orders
GROUP BY user_id;

-- MySQL: JSON_ARRAYAGG
SELECT
    user_id
    , JSON_ARRAYAGG(
        JSON_OBJECT(
            'order_id', id,
            'total', total,
            'status', status
        ) ORDER BY created_at DESC
    ) AS orders
FROM orders
GROUP BY user_id;
```

## Window Functions — Aggregates Without GROUP BY

```sql
-- Total revenue per user, on every order row
SELECT
    o.*
    , SUM(total) OVER (PARTITION BY user_id) AS user_revenue
    , AVG(total) OVER (PARTITION BY user_id) AS user_avg
    , COUNT(*) OVER (PARTITION BY user_id) AS user_orders
FROM orders o;

-- Running total
SELECT
    created_at
    , total
    , SUM(total) OVER (ORDER BY created_at ROWS UNBOUNDED PRECEDING) AS running_total
FROM orders
WHERE user_id = 1;

-- Moving average (3-order window)
SELECT
    created_at
    , total
    , AVG(total) OVER (
        ORDER BY created_at
        ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
    ) AS moving_avg_3
FROM orders;

-- Rank within partition
SELECT
    o.*
    , RANK() OVER (PARTITION BY user_id ORDER BY total DESC) AS order_rank
    , ROW_NUMBER() OVER (PARTITION BY user_id ORDER BY created_at) AS order_seq
FROM orders o;
```

---

## Frame Clauses (Window Functions)

| Frame | Meaning |
|-------|---------|
| `ROWS UNBOUNDED PRECEDING` | Start of partition to current row |
| `ROWS BETWEEN 2 PRECEDING AND CURRENT ROW` | Current + 2 previous |
| `ROWS BETWEEN UNBOUNDED PRECEDING AND UNBOUNDED FOLLOWING` | Entire partition |
| `RANGE` | Based on ORDER BY value (not row count) |

```sql
-- Default frame: RANGE UNBOUNDED PRECEDING AND CURRENT ROW
-- For ROWS: explicit is safer
SUM(total) OVER (
    PARTITION BY user_id
    ORDER BY created_at
    ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW
)
```

## Approximate Aggregates (Big Data)

```sql
-- PostgreSQL: HyperLogLog (approximate COUNT DISTINCT)
SELECT APPROX_COUNT_DISTINCT(user_id) FROM huge_events;

-- MySQL 8.0: APPROX_COUNT_DISTINCT
SELECT APPROX_COUNT_DISTINCT(user_id) FROM huge_events;

-- Use when exact count not required, table > 100M rows
-- Error rate: ~1-2%
```
